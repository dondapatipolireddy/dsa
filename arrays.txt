configuration without utilizing sort functions.

n = int(input())

low, mid, high = 0, 0, n - 1
while mid <= high:
if arr[mid] == 0:
arr[low], arr[mid] = arr[mid], arr[low]
low, mid = low + 1, mid + 1
elif arr[mid] == 1:
mid += 1
else:
arr[mid], arr[high] = arr[high], arr[mid]

class Solution:
    def longestPalindrome(self, s: str) -> str:
        max_=0
        start=0
        def expand(left,right):
            while left>=0 and right<len(s) and s[left]==s[right]:
                left-=1
                right+=1
            return left+1,right-left-1

        for i in range(len(s)):
            start1,l1=expand(i,i)
            start2,l2=expand(i,i+1)

            if max_<l1:
                start=start1
                max_=l1
            
            if max_<l2:
                start=start2
                max_=l2
        return s[start:start+max_]

class Solution:
    def subArrayRangesMax(self, arr):
        n = len(arr)

        # Previous Greater Element
        prev_greater = [-1] * n
        stack = []

        for i in range(n):
            while stack and arr[stack[-1]] <= arr[i]:
                stack.pop()
            prev_greater[i] = stack[-1] if stack else -1
            stack.append(i)

        # Next Greater Element
        next_greater = [n] * n
        stack = []

        for i in range(n - 1, -1, -1):
            while stack and arr[stack[-1]] < arr[i]:
                stack.pop()
            next_greater[i] = stack[-1] if stack else n
            stack.append(i)

        total = 0

        for i in range(n):
            left = i - prev_greater[i]
            right = next_greater[i] - i
            total += arr[i] * left * right

        return total

class Solution:

        n = len(arr)

        # Previous Smaller Element
        prev_smaller = [-1] * n
        stack = []

        for i in range(n):
            while stack and arr[stack[-1]] >= arr[i]:
                stack.pop()
            prev_smaller[i] = stack[-1] if stack else -1
            stack.append(i)

        # Next Smaller Element
        next_smaller = [n] * n
        stack = []

        for i in range(n - 1, -1, -1):
            while stack and arr[stack[-1]] > arr[i]:
                stack.pop()
            next_smaller[i] = stack[-1] if stack else n
            stack.append(i)

        MOD = 10**9 + 7
        total = 0

        for i in range(n):
            left = i - prev_smaller[i]
            right = next_smaller[i] - i
            total = (total + arr[i] * left * right) % MOD

        return total

class Solution:
    def subArrayRanges(self, arr):
        n = len(arr)

        # ---------------- MAX PART ----------------
        prev_greater = [-1] * n
        stack = []

        for i in range(n):
            while stack and arr[stack[-1]] <= arr[i]:
                stack.pop()
            prev_greater[i] = stack[-1] if stack else -1
            stack.append(i)

        next_greater = [n] * n
        stack = []

        for i in range(n - 1, -1, -1):
            while stack and arr[stack[-1]] < arr[i]:
                stack.pop()
            next_greater[i] = stack[-1] if stack else n
            stack.append(i)

        max_sum = 0
        for i in range(n):
            left = i - prev_greater[i]
            right = next_greater[i] - i
            max_sum += arr[i] * left * right

        # ---------------- MIN PART ----------------
        prev_smaller = [-1] * n
        stack = []

        for i in range(n):
            while stack and arr[stack[-1]] >= arr[i]:
                stack.pop()
            prev_smaller[i] = stack[-1] if stack else -1
            stack.append(i)

        next_smaller = [n] * n
        stack = []

        for i in range(n - 1, -1, -1):
            while stack and arr[stack[-1]] > arr[i]:
                stack.pop()
            next_smaller[i] = stack[-1] if stack else n
            stack.append(i)

        min_sum = 0
        for i in range(n):
            left = i - prev_smaller[i]
            right = next_smaller[i] - i
            min_sum += arr[i] * left * right

        return max_sum - min_sum

class Solution:
    def countSubstr (self, s, k):
        def atmost(k):
            d=dict()
            ans=0
            left=0
            for right in range(len(s)):
                d[s[right]]=d.get(s[right],0)+1
                while len(d)>k:
                    d[s[left]]-=1
                    if d[s[left]]==0:
                        del d[s[left]]
                    left+=1
                ans+=(right-left+1)
            return ans
        return atmost(k)-atmost(k-1)

class Solution:
    def partition(self, s: str) -> List[List[str]]:
        result=[]
        def p(a):
            return a==a[::-1]
        def palin(start,path):
            if start==len(s):
                result.append(path[:])
                return
            for i in range(start,len(s)):
                new=s[start:i+1]
                if p(new):
                    path.append(new)
                    palin(i+1,path)
                    path.pop()
        palin(0,[])
        return result

class Solution:
    def sortArray(self, nums: list[int]) -> list[int]:

        def mergesort(nums):                              
            if len(nums) <= 1:                                #important condition
                return nums

            mid = len(nums) // 2

            leftsub = mergesort(nums[:mid])
            rightsub = mergesort(nums[mid:])

            return merge(leftsub, rightsub)

        def merge(leftsub, rightsub):
            result = []
            i = j = 0

            while i < len(leftsub) and j < len(rightsub):
                if leftsub[i] <= rightsub[j]:
                    result.append(leftsub[i])
                    i += 1
                else:
                    result.append(rightsub[j])
                    j += 1

            result.extend(leftsub[i:])
            result.extend(rightsub[j:])

            return result

        return mergesort(nums)

 class Solution:

            if len(nums)<=1:                                      #important condition

            pivot=-1
            left=[]
            right=[]
            for i in range(len(nums)):
                if nums[i]<nums[pivot]:

                else:
                    right.append(nums[i])
