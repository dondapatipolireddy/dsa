class Solution:
    def pathSum(self, root: Optional[TreeNode], targetSum: int) -> List[List[int]]:
        a=[]
        def latest(root,pat):
            if not root:
                return
            if not root.left and not root.right:
                if targetSum==sum(pat):
                    a.append(pat.copy())
                return
            if root.left:
                pat.append(root.left.val)
                latest(root.left,pat)
                pat.pop()
            if root.right:
                pat.append(root.right.val)
                latest(root.right,pat)
                pat.pop()
        latest(root,[root.val])
        return a

class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        result=[]
        if not root:
            return result
        d=deque([root])
        while d:
            local=[]
            size=len(d)
            for i in range(size):
                node=d.popleft()
                local.append(node.val)
                if node.left:
                    d.append(node.left)
                if node.right:
                    d.append(node.right)
            result.append(local)
        return result

class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        result=[]
        if not root:
            return result
        d=deque([root])
        n=0
        while d:
            level=[]
            size=len(d)
            for i in range(size):
                node=d.popleft()
                level.append(node.val)
                if node.left:
                    d.append(node.left)
                if node.right:
                    d.append(node.right)
            if n%2==0:
                result.append(level)
            else:
                result.append(level[::-1])
            n+=1
        return result

class Solution:

        stack=[]
        def Tree(root,path):
            if not root:
                stack.append(path)
                return
            
            if not root.left and not root.right:       ***************************#important**********************************
                stack.append(path)
                return
            
            if root.left:
                temp=path+("->"+str(root.left.val))
                Tree(root.left,temp)
            
            if root.right:
                temp=path+("->"+str(root.right.val))
                Tree(root.right,temp)
        Tree(root,str(root.val))
        return stack

# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def flatten(self, root: Optional[TreeNode]) -> None:
        def link(root):
            if not root:
                return None
            left=root.left
            right=root.right
            lt=link(left)
            rt=link(right)
            if left:
                root.right=left
                root.left=None
                lt.right=right
            if rt:
                return rt
            if lt:
                return lt
            return root
        link(root)

class Solution:
    def search(self, root, key):
        if root.val == key:
            return root

        if root.val > key:
            return self.search(root.left, key)

        return self.search(root.right, key)

    def deleteNode(self, root: Optional[TreeNode], key: int) -> Optional[TreeNode]:

        if not root:
            return None

        rt = self.search(root, key)

        # No child
        if not rt.left and not rt.right:
            return None

        # Only right child
        if not rt.left:
            return rt.right

        # Only left child
        if not rt.right:
            return rt.left

        # Both children
        temp = rt.right

        while temp.left:
            temp = temp.left

        rt.val = temp.val

        rt.right = self.deleteNode(rt.right, temp.val)

        return root

class Solution:
    def isValidBST(self, root: TreeNode | None) -> bool:
        if not root:
            return True

        if root.right and root.val >= root.right.val:
            return False

        if root.left and root.val <= root.left.val:
            return False

        rr = self.isValidBST(root.right)
        ll = self.isValidBST(root.left)

        return ll and rr
