class Solution:
    def jobSequencing(self, deadline, profit):
        jobs = sorted(zip(profit, deadline), reverse=True)

        n = max(deadline)
        schedule = [-1] * (n + 1)

        count = 0
        total = 0

        for p, d in jobs:
            for j in range(min(d, n), 0, -1):
                if schedule[j] == -1:
                    schedule[j] = p
                    count += 1
                    total += p
                    break

        return [count, total]

class Solution:
    def knapsack(self, W, val, wt):
        n = len(val)
        dp = [[0] * (W + 1) for _ in range(n + 1)]

        for i in range(1, n + 1):
            for j in range(W + 1):
                if wt[i - 1] <= j:				***Important***
                    dp[i][j] = max(				***
                        val[i - 1] + dp[i - 1][j - wt[i - 1]],   **
                        dp[i - 1][j]				***
                    )						***
                else:						***
                    dp[i][j] = dp[i - 1][j]			***Important***

        return dp[n][W]  *****

class Solution:
    def fractionalKnapsack(self, val, wt, capacity):
        l=[]
        for val,wt in zip(val,wt):
            l.append([val/wt,val,wt])
        l.sort(reverse=True)
        profit=0
        for ratio,val,wt in l:
            if capacity>wt:
                capacity-=wt
                profit+=val
            else:
                profit+=((val*capacity)/wt)
                break
        return profit

class Solution:
    def canJump(self, nums: List[int]) -> bool:
        max_=nums[0]
        k=len(nums)
        count=1
        if max_==0 and k>1:
            return False
        for i in range(1,k):
            max_-=1
            count+=1
            if nums[i]>max_:
                max_=nums[i]
            if max_==0 and count==k:
                return True
            elif max_<=0:
                return False
        else:
            return True

class Solution:
    def merge(self, intervals):

        intervals.sort(key=lambda x: x[0])

        ans = [intervals[0]]

        for i in range(1, len(intervals)):

            if intervals[i][0] <= ans[-1][1]:
                ans[-1][1] = max(ans[-1][1], intervals[i][1])

            else:
                ans.append(intervals[i])

        return ans

class Solution:
    def eraseOverlapIntervals(self, intervals: List[List[int]]) -> int:
        count=0
        intervals.sort(key=lambda x:x[1])
        end=[intervals[0]]
        for s,e in intervals[1:]:
            if end[-1][1]>s:
                count+=1
                continue
            end.append([s,e])
        return count

2. Meeting Rooms II

The problem says:

A meeting can start exactly when another meeting ends.

Example

Meeting A : 9 - 10
Meeting B : 10 - 11

Room 1

9------10
        10------11

At exactly 10,

Meeting A ends.
Meeting B starts.

The same room can be reused.

So we free the room first.

class Solution:
    def minMeetingRooms(self,arr,dep):
        arr.sort()
        dep.sort()
        max_=0
        t=0
        j=0
        i=0
        while i<len(arr):
            if j>=len(dep):
                break
            if arr[i]<dep[j]:         important______________________________________
                t+=1
                max_=max(max_,t)
                i+=1
            else:
                t-=1
                j+=1
        return max_

minimum platforms:
_-----------------------------------------------------------------------------------------------------------------------------------
. Minimum Platforms (GFG)

The problem says:

A train arriving at the same time another train departs still needs another platform.

So if

Arrival = 10:00
Departure = 10:00

the departing train has not yet vacated the platform.

Platform 1

10:00 Departure Train A
10:00 Arrival   Train B

A is still occupying the platform.
B needs another platform.

Therefore,

if arrival <= departure:
    platforms += 1

because equal times are considered overlapping.

2. Meeting Rooms II

The problem says:

A meeting can start exactly when another meeting ends.

Example

Meeting A : 9 - 10
Meeting B : 10 - 11

Room 1

9------10
        10------11

At exactly 10,

Meeting A ends.
Meeting B starts.

The same room can be reused.

So we free the room first.

Hence

if start < end:
    rooms += 1
else:
    rooms -= 1

Notice the strict <.

When

start == end

we go to the else part, free a room, and then reuse it.

Visual comparison
Minimum Platforms
Train A

------10:00

Train B

      10:00------


Train A is still on the platform at 10:00.

Need

Platform 1 -> Train A
Platform 2 -> Train B

So

arrival <= departure
Meeting Rooms II
Meeting A

------10:00

Meeting B

      10:00------


Meeting A leaves exactly at 10:00.

Room becomes free instantly.

Meeting B enters.

Need only one room.

So

start < end
Interview Trick

Always read the statement.

If it says:

Equal time is allowed

then use

start < end

If it says:

Equal time still conflicts

then use

arrival <= departure
Easy rule to remember
Problem	Equal Time	Condition
Minimum Platforms	Conflict	arrival <= departure
Meeting Rooms II	No conflict	start < end
Meeting Rooms I	No conflict	current_end > next_start means overlap; current_end <= next_start is allowed
Merge Intervals	Touching intervals are usually merged if next_start <= current_end (depending on the problem definition)	

This difference is not about greedy algorithms—it's entirely determined by how the problem defines events that occur at the same time.

class Solution:
    def numRescueBoats(self, people: List[int], limit: int) -> int:
        people.sort()
        boat=0
        i=0
        j=len(people)-1
        while i<=j:
            if people[j]+people[i]>limit:
                j-=1
            else:
                i+=1
                j-=1
            boat+=1
        return boat
