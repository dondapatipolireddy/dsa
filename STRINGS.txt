class Solution:
    def characterReplacement(self, s: str, k: int) -> int:
        dic_=dict()
        left=0
        max_freq=0
        max_str=0

        for right in range(len(s)):
            dic_[s[right]]=dic_.get(s[right],0)+1
            if max_freq<dic_[s[right]]:
                max_freq=dic_[s[right]]
            while (right-left+1)-max_freq>k:
                dic_[s[left]]-=1
                left+=1
            max_str=max(max_str,(right-left+1))
        return max_str

from collections import deque,defaultdict,Counter
class Solution:
    def minWindow(self, s: str, t: str) -> str:
        dic=defaultdict(deque)
        dt=Counter(t)
        total_length=float('inf')
        start_length=-1
        for right,ch in enumerate(s):

            if s[right] not in t:
                continue

            dic[s[right]].append(right)
            valid=True    

            while len(dic[ch])>dt[ch]:            #update dict to new positions of t characters
                dic[ch].popleft()
     
            for ch in t:
                if len(dic[ch])<dt[ch]:
                    valid=False
                    break
            if valid:
                left=min(dic[ch][0] for ch in t)
                print(left)
                if (right-left+1)<total_length:
                    total_length=(right-left+1)
                    start_length=left
                    
        if start_length==-1:
            return ""
        return s[start_length:start_length+total_length]

class Solution:
    def validIPAddress(self, queryIP: str) -> str:

        # IPv4
        if queryIP.count(".") == 3:
            parts = queryIP.split(".")

            if len(parts) != 4:
                return "Neither"

            for part in parts:

                # Empty part
                if not part:
                    return "Neither"

                # Not digits
                if not part.isdigit():
                    return "Neither"

                # Leading zeros
                if len(part) > 1 and part[0] == '0':
                    return "Neither"

                # Range check
                if not (0 <= int(part) <= 255):
                    return "Neither"

            return "IPv4"

        # IPv6
        if queryIP.count(":") == 7:
            parts = queryIP.split(":")
            hexdigits = "0123456789abcdefABCDEF"

            if len(parts) != 8:
                return "Neither"

            for part in parts:

                # Length check
                if not (1 <= len(part) <= 4):
                    return "Neither"

                # Hexadecimal check
                for ch in part:
                    if ch not in hexdigits:
                        return "Neither"

            return "IPv6"

        return "Neither"
