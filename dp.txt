def knapsack(W, wt, val, n):

    dp = [[0]*(W+1) for _ in range(n+1)]

    for i in range(1,n+1):
        for j in range(W+1):

            if wt[i-1] <= j:

                dp[i][j] = max(
                    val[i-1] + dp[i-1][j-wt[i-1]],
                    dp[i-1][j]
                )

            else:
                dp[i][j] = dp[i-1][j]

    return dp[n][W]

class Solution:
    def generate(self, numRows: int) -> list[list[int]]:
        dp = []

        for i in range(numRows):
            row = [1] * (i + 1)

            for j in range(1, i):
                row[j] = dp[i-1][j-1] + dp[i-1][j]      #important

            dp.append(row)

        return dp

class Solution:
    def wordBreak(self, s: str, wordDict: list[str]) -> bool:
        n=len(s)
        d=[False]*(n+1)
        d[0]=True
        for i in range(1,n+1):
            for j in range(i):
                if d[j] and s[j:i] in wordDict:
                    d[i]=True
                    break
        return d[n]
# "I solve this problem using Dynamic Programming.

# I create a DP array where dp[i] represents whether the first i characters can be segmented into dictionary words.

# Initially, dp[0] = True.

# I use two loops to check substrings. If the previous part is valid (dp[j]) and the current substring (s[j:i]) exists in the dictionary, I mark dp[i] = True.

# Finally, I return dp[n]."

# Complexity:
# - Time: O(n² × substring cost).
# - Space: O(n) for DP.

class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        dp=[[0]*n for _ in range(m)]
        for i in range(m):
            dp[i][0]=1
        for i in range(n):
            dp[0][i]=1
        for i in range(1,m):
            for j in range(1,n):
                dp[i][j]=dp[i-1][j]+dp[i][j-1]
        return dp[m-1][n-1]

class Solution:
    def uniquePathsWithObstacles(self, obstacleGrid: list[list[int]]) -> int:
        m=len(obstacleGrid)
        n=len(obstacleGrid[0])
        dp=[[0]*n for _ in range(m)]
        for i in range(m):
            if obstacleGrid[i][0]!=1:
                if i>0:
                    if dp[i-1][0]!=0:
                        dp[i][0]=1
                else:
                    dp[i][0]=1
        for i in range(n):
            if obstacleGrid[0][i]!=1:
                if i>0:
                    if dp[0][i-1]!=0:
                        dp[0][i]=1
                else:
                    dp[0][i]=1

            for j in range(1,n):
                if obstacleGrid[i][j]!=1:
                    dp[i][j]=dp[i-1][j]+dp[i][j-1]
        return dp[m-1][n-1]

class Solution:
    def longestCommonSubsequence(self, text1: str, text2: str) -> int:
        m = len(text1)
        n = len(text2)

        dp = [[0]*(n+1) for _ in range(m+1)]

        for i in range(1,m+1):
            for j in range(1,n+1):

                if text1[i-1] == text2[j-1]:
                    dp[i][j] = 1 + dp[i-1][j-1]

                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])

        return dp[m][n]

class Solution:
    def longestPalindrome(self, s: str) -> str:
        max_=0
        start=-1
        def palindrome(left,right):
            while left>=0 and right<len(s) and s[left]==s[right]:
                left-=1
                right+=1
            return left+1,right-left-1
        for i in range(len(s)):
            if i+1<len(s):
                start1,l1=palindrome(i,i+1)
            start2,l2=palindrome(i,i)
            if i+1<len(s):
                if max_<l1:
                    start=start1
                    max_=l1
            if max_<l2:
                start=start2
                max_=l2
        return s[start:start+max_]
